// Jogo estilo Mario Bros com Macio e Camila
// Usando React Native + React Native Game Engine

import React, { useState, useRef, useEffect } from 'react';
import { View, Text, StyleSheet, Dimensions, TouchableOpacity } from 'react-native';
import { GameEngine } from 'react-native-game-engine';
import Sound from 'react-native-sound';

const { width, height } = Dimensions.get('window');

const Box = ({ position, color }) => {
  return (
    <View
      style={{
        position: 'absolute',
        width: 50,
        height: 50,
        backgroundColor: color,
        left: position[0],
        top: position[1],
      }}
    />
  );
};

const Obstacle = ({ position }) => {
  return (
    <View
      style={{
        position: 'absolute',
        width: 50,
        height: 50,
        backgroundColor: 'brown',
        left: position[0],
        top: position[1],
      }}
    />
  );
};

const gravity = 3;
const jumpForce = -30;
const screenBottom = height - 100;

const createLevel = (level) => {
  const baseObstacles = [
    { key: 'obst1', position: [300, screenBottom] },
    { key: 'obst2', position: [600, screenBottom] },
  ];

  if (level === 2) {
    baseObstacles.push({ key: 'obst3', position: [900, screenBottom] });
  } else if (level === 3) {
    baseObstacles.push(
      { key: 'obst3', position: [900, screenBottom] },
      { key: 'obst4', position: [1200, screenBottom] }
    );
  }

  const obstacles = {};
  baseObstacles.forEach(ob => {
    obstacles[ob.key] = { position: ob.position, renderer: <Obstacle /> };
  });

  return {
    player1: { position: [50, screenBottom], velocityY: 0, renderer: <Box color="blue" /> },
    player2: { position: [120, screenBottom], velocityY: 0, renderer: <Box color="pink" /> },
    ...obstacles
  };
};

const MoveSystem = (entities, { touches, dispatch }) => {
  const speed = 2;

  let player1 = entities.player1;
  let player2 = entities.player2;

  player1.position[0] += speed;
  player2.position[0] += speed;

  player1.velocityY += gravity;
  player2.velocityY += gravity;

  player1.position[1] += player1.velocityY;
  player2.position[1] += player2.velocityY;

  if (player1.position[1] > screenBottom) {
    player1.position[1] = screenBottom;
    player1.velocityY = 0;
  }
  if (player2.position[1] > screenBottom) {
    player2.position[1] = screenBottom;
    player2.velocityY = 0;
  }

  touches.filter(t => t.type === 'press').forEach(touch => {
    if (touch.event.pageX < width / 2 && player1.position[1] >= screenBottom) {
      player1.velocityY = jumpForce;
    }
    if (touch.event.pageX >= width / 2 && player2.position[1] >= screenBottom) {
      player2.velocityY = jumpForce;
    }
  });

  if (player1.position[0] > width * 2 || player2.position[0] > width * 2) {
    dispatch({ type: 'next-level' });
  }

  return entities;
};

export default function App() {
  const [running, setRunning] = useState(true);
  const [level, setLevel] = useState(1);
  const [entities, setEntities] = useState(createLevel(1));
  const [points, setPoints] = useState(0);
  const soundRef = useRef(null);

  useEffect(() => {
    Sound.setCategory('Playback');
    const sound = new Sound('background.mp3', Sound.MAIN_BUNDLE, (error) => {
      if (!error) {
        sound.setNumberOfLoops(-1);
        sound.play();
        soundRef.current = sound;
      }
    });

    return () => {
      if (soundRef.current) soundRef.current.release();
    };
  }, []);

  const handleEvent = (e) => {
    if (e.type === 'next-level') {
      const next = level + 1 > 3 ? 1 : level + 1;
      setLevel(next);
      setEntities(createLevel(next));
      setPoints(prev => prev + 10);
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Fase {level} - Macio & Camila</Text>
      <Text style={styles.points}>Pontos: {points}</Text>

      <GameEngine
        style={styles.gameContainer}
        running={running}
        systems={[MoveSystem]}
        onEvent={handleEvent}
        entities={entities}
      />

      <TouchableOpacity onPress={() => setRunning(!running)} style={styles.button}>
        <Text style={styles.buttonText}>{running ? 'Pausar' : 'Iniciar'}</Text>
      </TouchableOpacity>

      <Text style={styles.instructions}>Toque no lado esquerdo para Macio pular e no direito para Camila pular</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#a3d9ff',
    alignItems: 'center',
    justifyContent: 'center',
  },
  title: {
    fontSize: 24,
    fontWeight: 'bold',
    marginTop: 40,
  },
  points: {
    fontSize: 20,
    fontWeight: 'bold',
    color: '#2e2e2e',
    marginTop: 10,
  },
  gameContainer: {
    width: width,
    height: height * 0.6,
    backgroundColor: '#e0f7fa',
    marginTop: 20,
  },
  button: {
    marginTop: 20,
    padding: 10,
    backgroundColor: '#0077b6',
    borderRadius: 8,
  },
  buttonText: {
    color: 'white',
    fontSize: 18,
  },
  instructions: {
    marginTop: 10,
    fontSize: 16,
    color: '#333',
    paddingHorizontal: 20,
    textAlign: 'center',
  },
});
